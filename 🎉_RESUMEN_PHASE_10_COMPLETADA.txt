API ENDPOINT SUMMARY
================================================================================

1️⃣  POST /api/preparer-program/register
    Purpose: Register new preparer
    Auth: Public (rate limited)
    Rate limit: 5/hour
    Request body:
      {
        firstName, lastName, email, phone,
        businessName, practiceState, ein,
        yearsOfExperience, specializations,
        backgroundCheckConsent
      }
    Response: { preparerId, status: "pending_verification" }
    ✓ Queues background check job
    ✓ Encrypts PII (EIN)
    ✓ Creates preparer record

2️⃣  GET /api/preparer-program/profile
    Purpose: Get preparer profile
    Auth: Owner or Admin
    Query params: None
    Response: Full preparer object with licenses & renewals
    ✓ Decrypts sensitive fields
    ✓ Includes related data

3️⃣  PATCH /api/preparer-program/profile
    Purpose: Update preparer info
    Auth: Owner or Admin
    Request body: { businessName?, phone?, specializations? }
    Response: Updated preparer object
    ✓ Validates input
    ✓ Audit logs changes

4️⃣  POST /api/preparer-program/licenses/{preparerId}
    Purpose: Create new license (after background check)
    Auth: Admin only
    Request body: { certificationNumber, licenseType, issuedAt, expiryDate }
    Response: { license, preparer }
    ✓ Updates preparer status to "active"
    ✓ Sends welcome email
    ✓ Validates unique cert number

5️⃣  PATCH /api/preparer-program/licenses/{licenseId}
    Purpose: Update license status
    Auth: Admin only
    Request body: { status?, certificationNumber?, expiryDate? }
    Response: Updated license object
    ✓ Handles status transitions
    ✓ Validates date logic

6️⃣  POST /api/preparer-program/renewals
    Purpose: Submit license renewal
    Auth: Owner (preparer)
    Request body: { continuingEducationHours, confirmCompliance }
    Response: { renewal }
    ✓ Validates education hours (min 15)
    ✓ Checks license expiry soon
    ✓ Queues admin notification

7️⃣  GET /api/preparer-program/earnings?period=week|month|year
    Purpose: Get earnings statistics
    Auth: Owner or Admin
    Response: { earnings: { totalReturns, approvedReturns, totalEarnings, averageEarningsPerReturn } }
    ✓ Calculates from completed returns
    ✓ Supports multiple periods

8️⃣  GET /api/preparer-program/list?status=active&state=CA&page=1&limit=20
    Purpose: List all preparers (admin dashboard)
    Auth: Admin only
    Query params: status?, state?, search?, page, limit
    Response: { preparers: [], total, page, limit }
    ✓ Filtering by status & state
    ✓ Search by name/email/business
    ✓ Paginated results

DATABASE SCHEMA
================================================================================

Preparer {
  id: String
  tenantId: String → Tenant.id
  userId: String → User.id (unique 1:1)
  firstName, lastName, email, phone
  businessName, practiceState, ein (encrypted), specializations
  yearsOfExperience: Int
  status: enum (pending_verification, verified, active, suspended)
  backgroundCheckStatus: enum (pending, in_progress, completed, passed, failed)
  backgroundCheckResult: JSON
  verifiedAt?: DateTime
  licenses: PreparerLicense[]
  renewals: PreparerRenewal[]
  backgroundCheck: PreparerBackgroundCheck
  earnings: PreparerEarnings[]
  taxReturns: TaxReturn[]
  createdAt, updatedAt, deletedAt
}

PreparerLicense {
  id: String
  preparerId: String → Preparer.id (unique 1:1)
  certificationNumber: String (unique)
  status: enum (pending, active, expired, suspended, pending_renewal)
  licenseType: enum (CPA, EA, ENROLLED_AGENT)
  issuedAt: DateTime
  expiryDate: DateTime
  renewalDueDate?: DateTime
  lastRenewalDate?: DateTime
  renewalCount: Int
  createdAt, updatedAt
}

PreparerRenewal {
  id: String
  preparerId: String → Preparer.id
  licenseId: String (reference)
  continuingEducationHours: Int
  educationProof?: String (document ID)
  status: enum (pending_review, approved, rejected)
  reviewedBy?: String
  reviewedAt?: DateTime
  newExpiryDate?: DateTime
  submittedAt: DateTime
  createdAt, updatedAt
}

PreparerBackgroundCheck {
  id: String
  preparerId: String → Preparer.id (unique 1:1)
  status: enum (pending, in_progress, completed, passed, failed)
  checkProvider: String (checkr, sterling, etc.)
  externalCheckId?: String (from provider)
  result?: String (passed, failed, review_needed)
  resultDetails?: JSON (full report)
  criminalHistory: Boolean
  termsViolation: Boolean
  sanctionedList: Boolean
  requestedAt: DateTime
  completedAt?: DateTime
  expiresAt?: DateTime
  createdAt, updatedAt
}

PreparerEarnings {
  id: String
  preparerId: String → Preparer.id
  periodStart: DateTime
  periodEnd: DateTime
  period: String (day, week, month, year)
  totalReturns: Int
  approvedReturns: Int
  rejectedReturns: Int
  totalEarnings: Decimal
  platformFee: Decimal
  netEarnings: Decimal
  createdAt, updatedAt
  @@unique([preparerId, periodStart, periodEnd])
}
